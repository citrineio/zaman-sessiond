cmake_minimum_required(VERSION 3.21)
project(zaman_menu_presentation_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Quick)
add_executable(presentation-ui-test presentation-ui-test.cpp)
target_link_libraries(presentation-ui-test PRIVATE Qt6::Core Qt6::Gui Qt6::Quick)
target_compile_definitions(presentation-ui-test PRIVATE MENU_QML_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/../../qml")
enable_testing()
add_test(NAME menu-presentation-1x COMMAND presentation-ui-test)
add_test(NAME menu-presentation-2x COMMAND presentation-ui-test)
set_tests_properties(menu-presentation-1x PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_SCALE_FACTOR=1")
set_tests_properties(menu-presentation-2x PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_SCALE_FACTOR=2")
