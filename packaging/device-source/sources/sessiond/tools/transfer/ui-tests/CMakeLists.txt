cmake_minimum_required(VERSION 3.21)
project(zaman_transfer_ui LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Quick)
add_executable(transfer-ui-test transfer-ui-test.cpp)
target_link_libraries(transfer-ui-test PRIVATE Qt6::Core Qt6::Gui Qt6::Quick)
enable_testing()
add_test(NAME transfer-ui-1x COMMAND transfer-ui-test "${MENU_QML_ROOT}")
add_test(NAME transfer-ui-2x COMMAND transfer-ui-test "${MENU_QML_ROOT}")
set_tests_properties(transfer-ui-1x PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_SCALE_FACTOR=1")
set_tests_properties(transfer-ui-2x PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_SCALE_FACTOR=2")
